-- Feature 2: Book checkout / check-in lifecycle
-- New in 1.57.00-hardened.1

CREATE TABLE `planning_loan` (
  `loan_id` INT NOT NULL AUTO_INCREMENT,
  `resource_id` VARCHAR(20) COLLATE latin1_general_ci NOT NULL,
  `user_id` VARCHAR(20) COLLATE latin1_general_ci NULL DEFAULT NULL,
  `borrower_name` VARCHAR(100) COLLATE latin1_general_ci NULL DEFAULT NULL,
  `date_out` DATE NOT NULL,
  `date_due` DATE NULL DEFAULT NULL,
  `date_in` DATE NULL DEFAULT NULL,
  `statut` ENUM('out','returned','lost') COLLATE latin1_general_ci NOT NULL DEFAULT 'out',
  `note` VARCHAR(255) COLLATE latin1_general_ci NULL DEFAULT NULL,
  `reminded_at` DATETIME NULL DEFAULT NULL,
  PRIMARY KEY (`loan_id`),
  KEY `idx_loan_resource` (`resource_id`),
  KEY `idx_loan_user` (`user_id`),
  KEY `idx_loan_statut` (`statut`),
  CONSTRAINT `fk_loan_resource` FOREIGN KEY (`resource_id`) REFERENCES `planning_ressource` (`ressource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_loan_user` FOREIGN KEY (`user_id`) REFERENCES `planning_user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=latin1 COLLATE=latin1_general_ci;

CREATE TABLE `planning_loan_hold` (
  `hold_id` INT NOT NULL AUTO_INCREMENT,
  `resource_id` VARCHAR(20) COLLATE latin1_general_ci NOT NULL,
  `user_id` VARCHAR(20) COLLATE latin1_general_ci NULL DEFAULT NULL,
  `borrower_name` VARCHAR(100) COLLATE latin1_general_ci NULL DEFAULT NULL,
  `requested_at` DATETIME NOT NULL,
  `statut` ENUM('waiting','notified','fulfilled','cancelled') COLLATE latin1_general_ci NOT NULL DEFAULT 'waiting',
  PRIMARY KEY (`hold_id`),
  KEY `idx_hold_resource` (`resource_id`),
  CONSTRAINT `fk_hold_resource` FOREIGN KEY (`resource_id`) REFERENCES `planning_ressource` (`ressource_id`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_hold_user` FOREIGN KEY (`user_id`) REFERENCES `planning_user` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=latin1 COLLATE=latin1_general_ci;
