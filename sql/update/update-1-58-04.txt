-- Site-configurable supervisor levels. Each site defines its own level names
-- and capabilities (lead a group / cover individual / be assigned a trainee).
-- Seeded with the HealthPoint BHC defaults; codes match planning_ressource.niveau
-- so existing supervisors keep their level. New in 1.58.00.

CREATE TABLE `planning_supervisor_level` (
  `level_id` INT NOT NULL AUTO_INCREMENT,
  `code` VARCHAR(20) COLLATE latin1_general_ci NOT NULL,
  `label` VARCHAR(60) COLLATE latin1_general_ci NOT NULL,
  `ordre` INT NOT NULL DEFAULT 0,
  `lead_group` ENUM('oui','non') COLLATE latin1_general_ci NOT NULL DEFAULT 'oui',
  `cover` ENUM('oui','non') COLLATE latin1_general_ci NOT NULL DEFAULT 'non',
  `assignable` ENUM('oui','non') COLLATE latin1_general_ci NOT NULL DEFAULT 'non',
  PRIMARY KEY (`level_id`),
  UNIQUE KEY `uq_level_code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=latin1 COLLATE=latin1_general_ci;
